class General(object):
    def copy_to(self, target: Self, deep: bool = False) -> None:
        if type(self) is not type(target):
            raise TypeError("Объекты должны иметь одинаковый тип")

        if deep:
            target.__dict__ = copy.deepcopy(self.__dict__)
        else:
            target.__dict__ = copy.copy(self.__dict__)


    def clone(self, deep: bool = True) -> Self:
        if deep:
            return copy.deepcopy(self)
        return copy.copy(self)


    def equals(self, other: object) -> bool:
        return (
            type(self) is type(other)
            and self.__dict__ == other.__dict__
        )


    def serialize(self) -> str:
        data = pickle.dumps(self)
        return base64.b64encode(data).decode("ascii")


    @classmethod
    def deserialize(cls, data: str) -> Self:
        raw = base64.b64decode(data.encode("ascii"))
        obj = pickle.loads(raw)

        if not isinstance(obj, cls):
            raise TypeError("Десериализован объект другого типа")

        return obj


    def __str__(self) -> str:
        return f"{type(self).__name__}({self.__dict__})"


    def is_type(self, cls: type) -> bool:
        return isinstance(self, cls)


    def get_type(self) -> type:
        return type(self)


class Any(General):
    pass
